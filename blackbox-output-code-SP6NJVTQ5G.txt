'use client';
import { useSession, signOut } from 'next-auth/react';
import Link from 'next/link';
import { useState } from 'react';

export default function Navbar() {
  const { data: session } = useSession();
  const [isOpen, setIsOpen] = useState(false);

  return (
    <nav className="bg-gray-800 sticky top-0 z-50 shadow-lg">
      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
        <div className="flex justify-between h-16">
          <div className="flex items-center">
            <Link href="/" className="text-2xl font-bold text-purple-400">NEXORA</Link>
          </div>
          <div className="hidden md:flex items-center space-x-4">
            <Link href="/" className="hover:text-purple-400">Home</Link>
            <Link href="/marketplace" className="hover:text-purple-400">Marketplace</Link>
            {session && <Link href="/dashboard" className="hover:text-purple-400">Dashboard</Link>}
            {!session ? (
              <>
                <Link href="/auth/signin" className="hover:text-purple-400">Login</Link>
                <Link href="/auth/signup" className="hover:text-purple-400">Register</Link>
              </>
            ) : (
              <button onClick={() => signOut()} className="hover:text-purple-400">Logout</button>
            )}
          </div>
          <div className="md:hidden">
            <button onClick={() => setIsOpen(!isOpen)} className="text-white">
              ☰
            </button>
          </div>
        </div>
        {isOpen && (
          <div className="md:hidden">
            <Link href="/" className="block py-2">Home</Link>
            <Link href="/marketplace" className="block py-2">Marketplace</Link>
            {session && <Link href="/dashboard" className="block py-2">Dashboard</Link>}
            {!session ? (
              <>
                <Link href="/auth/signin" className="block py-2">Login</Link>
                <Link href="/auth/signup" className="block py-2">Register</Link>
              </>
            ) : (
              <button onClick={() => signOut()} className="block py-2">Logout</button>
            )}
          </div>
        )}
      </div>
    </nav>
  );
}
